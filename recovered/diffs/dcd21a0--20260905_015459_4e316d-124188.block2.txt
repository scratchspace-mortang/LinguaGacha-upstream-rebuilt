    21	/** 解码 data: URL 为字节；仅处理 base64，否则让上游处理。 */
    22	function decode_data_url_to_bytes(href: string): ArrayBuffer {
    23	  const comma = href.indexOf(",");
    24	  if (comma === -1) throw new Error("Invalid data URL.");
    25	  const meta = href.slice(5, comma); // "data:<mime>;base64"
    26	  if (!meta.endsWith(";base64")) throw new Error("Only base64 data URLs are decoded.");
    27	  const payload = href.slice(comma + 1);
    28	  const buffer = Buffer.from(payload, "base64");
    29	  // 拷贝到独立 ArrayBuffer，Buffer 池化的内存可能被回收
    30	  const out = new ArrayBuffer(buffer.byteLength);
    31	  new Uint8Array(out).set(buffer);
    32	  return out;
    33	}

    35	/** 读取 data URL 的 MIME；缺省走 text/plain;charset=US-ASCII 与 RFC 2397 一致。 */
    36	function read_data_url_mime(href: string): string {
    37	  const comma = href.indexOf(",");
    38	  if (comma === -1) return "text/plain;charset=US-ASCII";
    39	  const meta = href.slice(5, comma); // "data:<mime>" 或 "data:<mime>;base64"
    40	  const semi = meta.indexOf(";");
    41	  const mime = (semi === -1 ? meta : meta.slice(0, semi)).trim();
    42	  return mime === "" ? "text/plain;charset=US-ASCII" : mime;

    63	  /** 把跨 realm 的 Response（undici / Node 原生 fetch）重新包成本线程 Response，使
    64	   * 消费方 `instanceof Response` 检查通过。data: URL 在 Electron worker 里
    65	   * 是 harper.js 唯一会触发的非 http(s) 协议，跨 realm Response 类差异会
    66	   * 导致 wasm 加载路径进入 `WebAssembly.instantiate(<Response>, imports)`
    67	   * 而撞 "Argument 0 must be a buffer source"。 */
    68	  private async rewrap_response_to_local_realm(r: Response): Promise<Response> {
    69	    if (this.previous_response_ctor === null) return r;
    70	    if (r instanceof this.previous_response_ctor) return r;
    71	    const source = r as unknown as Response; // unknown realm Response: widen type for property access
    72	    proxy_fetch_log("fetch.rewrap_response", {
    73	      ctor_name: source?.constructor?.name ?? "unknown",
    74	      status: source.status,
    75	    });

    77	    // 用源 Headers 构造新 Headers 实例，避免 Node 24 Response 构造器在传入 Body 时
    78	    // 自动补 Content-Type 覆盖源 headers 的语义。

    80	    for (const [name, value] of source.headers.entries()) headers.append(name, value);
    81	    return new this.previous_response_ctor(body, {
    82	      status: source.status,
    83	      statusText: source.statusText,
    84	      headers,
    85	    });
    86	  }

    88	  /** 普通远端 HTTP 共用此 Fetch；专用安全下载链路仍显式选择自己的 dispatcher。
    89	   * 非 http(s) 协议（data:、blob:、file:）委托给安装期保存的线程原 fetch：
    90	   * undici 的 Response 与线程原 Response 类不互通（`instanceof Response` 跨类失败），
    91	   * 消费方拿到的 Response 会落到 `WebAssembly.instantiate(<Response>, imports)`
    92	   * 撞 "Argument 0 must be a buffer source"。本类只接管真正走代理的远程 HTTP。 */
    93	  public readonly fetch: typeof globalThis.fetch = async (input, init) => {
    94	    if (this.disposed) {
    95	      throw new Error("System proxy HTTP client is disposed.");
    96	    }
    97	    const url = read_request_url(input);
    98	    proxy_fetch_log("fetch.entered", url_diagnostic(url));
    99	    if (url.protocol !== "http:" && url.protocol !== "https:") {
   100	      // data: URL 在 Electron worker_thread 里撞 harper.js wasm glue 的
   101	      // `instanceof Response` 跨模块判定歧义：glue 闭包里的 Response 与本线程
   102	      // 的 Response 是不同的标识符，即使 `instanceof` 同源也会让 harper.js
   103	      // 退到 `WebAssembly.instantiate(<Response>, imports)` 分支而抛
   104	      // "Argument 0 must be a buffer source"。
   105	      //
   106	      // 选项 B：harper.js 的 getInitInput 已直接解码 data: URL 为 Uint8Array
   107	      // 并交给 WebAssembly.instantiate，这条 Response 路径不再被 wasm-bindgen
   108	      // 触达；保留 data: 提前返回 Response 是为了让非 wasm 的调用方
   109	      // （如测试）拿到带正确 Content-Type 的 Response。
   110	      if (url.protocol === "data:") {
   111	        const bytes = decode_data_url_to_bytes(url.href);
   112	        const mime = read_data_url_mime(url.href);
   113	        return new this.previous_response_ctor!(bytes, {
   114	          status: 200,
   115	          statusText: "OK",
   116	          headers: new Headers({ "Content-Type": mime }),
   117	        });
   118	      }
   119	      proxy_fetch_log("fetch.delegate_previous", url_diagnostic(url));
   120	      if (this.previous_fetch === null) {
   121	        throw new Error(`System proxy HTTP client has no previous fetch for ${url.protocol}.`);
   122	      }
   123	      const upstream = await this.previous_fetch(
   124	        input as Parameters<typeof globalThis.fetch>[0],
   125	        init,
   126	      );
   127	      return await this.rewrap_response_to_local_realm(upstream);
   128	    }
   129	    const signal = init?.signal ?? undefined;
   130	    const dispatcher = is_loopback_hostname(url.hostname)
   131	      ? this.direct_dispatcher
   132	      : this.resolve_dispatcher(
   133	          parse_system_proxy_route(await this.resolver.resolveProxy(url.href, signal)),
   134	        );
   135	    proxy_fetch_log("fetch.dispatch_undici", {
   136	      protocol: url.protocol,
   137	      hostname: url.hostname,
   138	      loopback: is_loopback_hostname(url.hostname),
   139	    });
   140	    return (await undici_fetch(input as Parameters<typeof undici_fetch>[0], {
   141	      ...(init as Parameters<typeof undici_fetch>[1]),
   142	      dispatcher,
   143	      redirect: "error",
   144	    })) as unknown as Response;
   145	  };

   147	  /** Backend 线程只有一个普通 HTTP transport，第三方 SDK 也从同一全局入口取用。 */
   148	  public install_as_global_fetch(): void {
   149	    if (this.disposed || this.previous_fetch !== null) {
   150	      throw new Error("System proxy HTTP client cannot be installed in its current state.");
   151	    }
   152	    this.previous_fetch = globalThis.fetch;
   153	    // 捕获安装期线程原 Response 构造器，供 data: URL 跨 realm Response 重新包装用。
   154	    this.previous_response_ctor = globalThis.Response;
   155	    globalThis.fetch = this.fetch;
   156	  }

   158	  /** 先恢复线程原 transport，再关闭本 Client 创建的全部连接池。 */
   159	  public async dispose(): Promise<void> {
   160	    if (this.disposed) return;
   161	    this.disposed = true;
   162	    if (this.previous_fetch !== null) {
   163	      globalThis.fetch = this.previous_fetch;
   164	      this.previous_fetch = null;
   165	    }
   166	    this.previous_response_ctor = null;
   167	    const dispatchers = [this.direct_dispatcher, ...this.proxy_dispatchers.values()];
   168	    this.proxy_dispatchers.clear();
   169	    const results = await Promise.allSettled(dispatchers.map((dispatcher) => dispatcher.close()));
   170	    const errors = results
   171	      .filter((result): result is PromiseRejectedResult => result.status === "rejected")
   172	      .map((result) => result.reason);
   173	    if (errors.length > 0) {
   174	      throw new AggregateError(errors, "System proxy HTTP client failed to close.");
   175	    }
   176	  }

   178	  /** 按已解析路由返回直连池，或复用对应协议与地址的代理池。 */
   179	  private resolve_dispatcher(route: SystemProxyRoute): Dispatcher {
   180	    if (route.kind === "direct") return this.direct_dispatcher;
   181	    const key = `${route.kind}:${route.uri}`;
   182	    const existing = this.proxy_dispatchers.get(key);
   183	    if (existing !== undefined) return existing;
   184	    const dispatcher =
   185	      route.kind === "socks5" ? new Socks5ProxyAgent(route.uri) : new ProxyAgent(route.uri);
   186	    this.proxy_dispatchers.set(key, dispatcher);
   187	    return dispatcher;
   188	  }
   189	}

   191	/** 按 Chromium 返回顺序选择首个可执行路由；只有显式 DIRECT 才允许直连。 */
   192	export function parse_system_proxy_route(proxy_rules: string): SystemProxyRoute {
   193	  for (const rule of proxy_rules.split(";")) {
   194	    const trimmed_rule = rule.trim();
   195	    if (trimmed_rule === "") continue;
   196	    const [raw_type = "", ...target_parts] = trimmed_rule.split(/\s+/u);
   197	    const type = raw_type.toUpperCase();
   198	    if (type === "DIRECT") return { kind: "direct" };
   199	    const target = target_parts.join(" ");
   200	    if (type === "PROXY") {
   201	      const uri = build_proxy_uri("http", target);
   202	      if (uri !== null) return { kind: "proxy", uri };
   203	    }
   204	    if (type === "HTTPS") {
   205	      const uri = build_proxy_uri("https", target);
   206	      if (uri !== null) return { kind: "proxy", uri };
   207	    }
   208	    if (type === "SOCKS" || type === "SOCKS5") {
   209	      const uri = build_proxy_uri("socks5", target);
   210	      if (uri !== null) return { kind: "socks5", uri };
   211	    }
   212	  }
   213	  throw new Error("System proxy returned no supported route.");
   214	}

   222	/**
   223	 * 拼装代理诊断输出，避免把查询字符串与 userinfo 写进 stderr：
   224	 * 自定义模型端点常用 query 携带 API key 与签名，data: URL 的 base64 payload
   225	 * 也可能携带内容。诊断需要的只是路由信息（协议 + 主机），其他字段省略。
   226	 * data: URL 的 meta 段（mimetype 与可选 ;base64 等）按
   227	 * MAX_DATA_URL_DIAGNOSTIC_PREFIX 截断，避免用户控制的长 meta
   228	 * 把任意字节塞进 stderr；无逗号的非法 data URL 不暴露原文。
   229	 */
   230	const MAX_DATA_URL_DIAGNOSTIC_PREFIX = 256;
   231	const INVALID_DATA_URL_HEAD = "<invalid>";
   232	
   233	function url_diagnostic(url: URL): Record<string, unknown> {
   234	  if (url.protocol === "data:") {
   235	    const comma = url.href.indexOf(",");
   236	    if (comma === -1) {
   237	      return { protocol: url.protocol, data_head: INVALID_DATA_URL_HEAD };
   238	    }
   239	    const raw_head = url.href.slice(0, comma);
   240	    let data_head: string;
   241	    if (raw_head.length <= MAX_DATA_URL_DIAGNOSTIC_PREFIX) {
   242	      data_head = raw_head;
   243	    } else {
   244	      // 留出 1 字符给省略号，保证截断后总长严格不超过 MAX_DATA_URL_DIAGNOSTIC_PREFIX。
   245	      const TRUNCATION_MARKER = "…";
   246	      const keep = MAX_DATA_URL_DIAGNOSTIC_PREFIX - TRUNCATION_MARKER.length;
   247	      data_head = `${raw_head.slice(0, keep)}${TRUNCATION_MARKER}`;
   248	    }
   249	    return { protocol: url.protocol, data_head };
   250	  }
   251	  return { protocol: url.protocol, hostname: url.hostname };
   252	}

   254	/** Chromium 代理目标通常省略 scheme；无效目标留给后续候选继续匹配。 */
   255	function build_proxy_uri(protocol: "http" | "https" | "socks5", target: string): string | null {
   256	  const trimmed_target = target.trim();
   257	  if (trimmed_target === "") return null;
   258	  try {
   259	    const url = new URL(
   260	      trimmed_target.includes("://") ? trimmed_target : `${protocol}://${trimmed_target}`,
   261	    );
   262	    return url.hostname === "" ? null : url.toString();
   263	  } catch {
   264	    return null;
   265	  }
   266	}

   268	/** 本机模型端点不经过外部代理，覆盖 localhost、IPv6 loopback 与完整 127/8。 */
   269	function is_loopback_hostname(hostname: string): boolean {
   270	  const normalized = hostname.toLowerCase();
   271	  return (
   272	    normalized === "localhost" ||
   273	    normalized === "[::1]" ||
   274	    normalized === "::1" ||
   275	    normalized.startsWith("127.")
   276	  );
   277	}
